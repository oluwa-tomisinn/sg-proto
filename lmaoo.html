def interpolate_color(hex1, hex2, steps): c1 = [int(hex1[i:i+2], 16) for i in (1, 3, 5)] c2 = [int(hex2[i:i+2], 16) for i in (1, 3, 5)] palette = [] for i in range(steps): t = i / (steps - 1) curr = [round(c1[j] + (c2[j] - c1[j]) * t) for j in range(3)] palette.append(f"#{curr[0]:02X}{curr[1]:02X}{curr[2]:02X}") return palette print("5 steps:", interpolate_color("#0373BF", "#FFFFFF", 5)) print("7 steps:", interpolate_color("#0373BF", "#FFFFFF", 7))
